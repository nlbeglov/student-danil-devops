> [!tip]
> Скрипт, на который по хорошему заменить создание пользователей представлен ниже
> 
> Скрипт нужен для того чтобы в более удобном (и правильном) формате хранить имена пользователь - пароль. 

```
#!/usr/bin/env bash
set -euo pipefail

USERNAME="${1:?Использование: create-user.sh <username> <email> [--admin]}"
EMAIL="${2:?Укажи email}"
EXTRA_FLAG="${3:-}"

OUTPUT=$(docker compose exec -u git server gitea admin user create \
  --username "$USERNAME" --email "$EMAIL" --random-password $EXTRA_FLAG)

# ищем пароль внутри одинарных кавычек на строке, где встречается слово "password" —
# не привязываемся к точной формулировке CLI (она может отличаться между версиями Gitea)
PASSWORD=$(echo "$OUTPUT" | grep -i password | grep -oE "'[^']+'" | tr -d "'" | tail -1)

echo "$USERNAME password: $PASSWORD" >> /opt/task-01/secrets/credentials.txt
chmod u=rw,go= /opt/task-01/secrets/credentials.txt

echo "Пользователь $USERNAME создан, пароль сохранён в credentials.txt"
SCRIPT
chmod +x /opt/task-01/scripts/create-user.sh
```
## 2. Конфигурации
 Файл docker-compose.yml
```                                                        
networks:
  gitea:
    external: false

services:
  server:
    image: docker.gitea.com/gitea:1.27.3
    container_name: gitea
    environment:
      - USER_UID=1000
      - USER_GID=1000
      - GITEA__database__DB_TYPE=postgres
      - GITEA__database__HOST=db:5432
      - GITEA__database__NAME=${POSTGRES_DB}
      - GITEA__database__USER=${POSTGRES_USER}
      - GITEA__database__PASSWD=${POSTGRES_PASSWORD}
      - GITEA__service__DISABLE_REGISTRATION=true
      - GITEA__server__DOMAIN=${DOMAIN}
      - GITEA__server__ROOT_URL=https://${DOMAIN}/
      - GITEA__security__INSTALL_LOCK=true
      - GITEA__DEFAULT__APP_NAME=PolitBlocks
    restart: always
    networks:
      - gitea
    volumes:
      - ./gitea:/data
      - /etc/timezone:/etc/timezone:ro
      - /etc/localtime:/etc/localtime:ro
    depends_on:
      - db

  db:
    image: docker.io/library/postgres:14
    restart: always
    environment:
      - POSTGRES_USER=${POSTGRES_USER}
      - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
      - POSTGRES_DB=${POSTGRES_DB}
    networks:
      - gitea
    volumes:
      - ./postgres:/var/lib/postgresql/data

  caddy:
    image: caddy:2.11.4
    container_name: caddy
    restart: always
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./Caddyfile:/etc/caddy/Caddyfile
      - caddy_data:/data
      - caddy_config:/config
    networks:
      - gitea

volumes:
  caddy_data:
  caddy_config:
```
