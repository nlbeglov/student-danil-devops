# Требование 4
### Был подготовлен полный скрипт работы с комментариями
``` 
mkdir -p /opt/task-01/backup
cd /opt/task-01
# останавливаем gitea 
docker compose stop server
# снимаем дамп бд
docker compose exec -T db pg_dump -U gitea -d gitea -F c > /opt/task-01/backup/gitea_db.dump
# архивируем файлы gitea
tar -czf /opt/task-01/backup/gitea-data.tar.gz -C /opt/task-01 gitea
# создаем безопасный .env без сикретов
sed -E 's/=.*/=CHANGE_ME/' .env > .env.example
# копируем другие важные файлы в архив
cp docker-compose.yml Caddyfile .env.example /opt/task-01/backup/
# запускаем gitea
docker compose start server
# подготавливаем и безопасим архив.
# генерируем имя файла и пароль 
BACKUP_FILE="backup-$(date +%F_%H%M).tar.gz"
BACKUP_PASSPHRASE=$(openssl rand -base64 24)
# записываем пароль от бэкапа в credentials.txt
echo "Backup passphrase ($BACKUP_FILE): $BACKUP_PASSPHRASE" >> /opt/task-01/credentials.txt
cd /opt/task-01
# создаем основной архив
tar -czf "$BACKUP_FILE" backup/
# паролим архив
echo "$BACKUP_PASSPHRASE" | gpg --batch --yes --pinentry-mode loopback --passphrase-fd 0 -c "$BACKUP_FILE"
# удаляем мусор 
rm "$BACKUP_FILE"
rm /opt/task-01/backup
unset BACKUP_PASSPHRASE
unset BACKUP_FILE
```
